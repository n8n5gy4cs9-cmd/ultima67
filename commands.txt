# Ultima67 console commands. GENERATED from the registry:
#   cargo run -q -p u67_console --bin gen_commands > commands.txt
# Open the console with ` or F1; cheats menu with F2. Toggles: "cmd [on|off]".
# Format: command <required> [optional] - description

# --- console ---
help [command]                    - list commands / show usage
cmds                              - print all commands
clear                             - clear console output
echo <text>                       - print text
exec <file>                       - run a script of commands

# --- items ---
give <item_id> [count]            - add item to backpack (give vainamoinen_gun)
give_all_weapons                  - add one of every weapon
give_all_items                    - add one of every item
take <item_id> [count]            - remove item
drop_all                          - drop inventory on ground
ammo [count]                      - refill ammo for equipped gun
money <amount>                    - add silver
items [filter]                    - list item ids

# --- player ---
god [on|off]                      - invulnerable
heal [amount]                     - heal player (full if omitted)
mana [amount]                     - restore mana (väki)
infinite_ammo [on|off]            - never reload/consume ammo
noclip [on|off]                   - walk through walls
fast [on|off]                     - 3x move speed
xp <amount>                       - add experience
level <n>                         - set level
stat <str|dex|int|vaki> <n>       - set stat
skill <name> <n>                  - set skill
kill_all                          - kill all hostile NPCs nearby
invisible [on|off]                - enemies ignore you
one_hit [on|off]                  - all attacks kill

# --- world ---
tp <place_id>                     - teleport to named place (tp kaupang, tp mimir_well)
tpxy <x> <y>                      - teleport to coordinates on current map
pos                               - print position
planet <body_id>                  - travel to planet (planet maani, planet pohjola)
planets                           - list bodies
places [filter]                   - list teleport ids
reveal_map                        - reveal full map
unlock_travel                     - unlock all fast-travel nodes
time <HH:MM>                      - set game time
timescale <x>                     - game time speed multiplier
weather <clear|rain|snow|storm|aurora>- set weather

# --- spawn ---
spawn <npc_or_creature_id> [count]- spawn at cursor
spawn_ship <ship_id>              - spawn ship
spawn_cannon                      - spawn a cannon at cursor
creatures [filter]                - list spawnable ids

# --- magic ---
spells_all                        - learn all spells
reagents                          - fill reagents
cast <spell_id>                   - cast without cost

# --- quests ---
quest <list|start|complete|reset> [id]- list quests / change quest state

party <add|remove> <npc_id>       - party management
ending <a|b|c>                    - jump to ending (debug)

# --- game ---
save [slot]                       - save
load [slot]                       - load
quit                              - quit game
splitscreen <players 1-4>         - set local players (debug)

# --- dev ---
fps [on|off]                      - show fps

reload_assets                     - hot-reload assets
reload_data                       - reload data files
screenshot                        - save screenshot

set <cvar> <value>                - set config variable
get <cvar>                        - read config variable
